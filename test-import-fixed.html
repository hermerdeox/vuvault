Additionally, we now ensure a master key exists before importing:

// Check and create master key if needed let masterKey = await getMasterKey(); if (!masterKey) { const defaultKey = window.crypto.getRandomValues(new Uint8Array(32)) .reduce((acc, byte) => acc + byte.toString(16).padStart(2, '0'), ''); await setMasterKey(defaultKey); }
